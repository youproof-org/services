[latex display="true"]\begin{aligned}x&\ eq y\\v_p(xy)&=0\\v_p(x-y)&\ eq 0\end{aligned}[/latex]

[latex display="true"]v_p(x^p-y^p)=v_p(x-y)+1[/latex]
